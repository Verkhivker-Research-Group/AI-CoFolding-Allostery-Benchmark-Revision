SECTION 0: KEY DIFFERENCES FROM THE MAIN (ASD) PIPELINE
================================================================================

The main pipeline uses ASD-format AF3 folders:
  ASD00060000_1_3HRF_P47_r2
  └─ parts[2] = PDB ID  (3HRF)   ← used for reference lookup
  └─ parts[3] = ligand code (P47)

The PLA AF3 folders use a DIFFERENT naming convention:
  summary_B6YWB8_N1_B6YWB8_DTP_r97          (single-underscore delimited)
  └─ parts[0] = "summary"          ← PLA identifier prefix
  └─ parts[1] = UniProt ID (B6YWB8) ← NOT a PDB ID
  └─ parts[2] = Nn binding site #  (N1, N2, ...)
  └─ parts[3] = UniProt ID again   (same as parts[1])
  └─ parts[4] = ligand code (DTP)
  └─ parts[5] = run # (r97)

  compound_key = f"{parts[1]}_{parts[4]}".lower()  →  "b6ywb8_dtp"

The PLA Chai folders use DOUBLE-underscore delimiters and INCLUDE the PDB code:
  B6YWB8_N1__6KBD__DTP                       (double-underscore delimited)
  └─ part[0] (split by "__") = "B6YWB8_N1"   → UniProt=B6YWB8, site=N1
  └─ part[1] = PDB ID (6KBD)                 ← PDB code IS embedded!
  └─ part[2] = ligand code (DTP)

  compound_key = f"{uniprot}_{lig}".lower()  →  "b6ywb8_dtp"
  (same key as AF3 PLA — enables cross-producer comparison)

KEY DIFFERENCES vs main pipeline:
  1. AF3 PLA: UniProt ID in folder name (not PDB) → needs UniProt→PDB mapping
             for reference CIF download
  2. Chai PLA: PDB ID IS in folder name → direct reference lookup, no mapping
  3. Both: compound_key = {uniprot}_{lig} (not {pdb}_{lig} like ASD)
  4. Both: id column in output CSV = UniProt ID (not 4-char PDB)
  5. Chai PLA: multiple PDB structures per UniProt+LIG are NORMAL
     (e.g., D9J2T9+LP5 has 3QJI, 4EMR, 4GUW → 3×5 = 15 models for that key)
     The "best" selection picks the single model with lowest pose RMSD across all.

  import requests, csv, time
  from pathlib import Path

  uniprot_ids = Path("/tmp/pla_uniprots.txt").read_text().split()
  out = Path("/mnt/c/Users/Ryan/AI-CoFolding-Allostery-Benchmark/references/pla_uniprot_map.csv")

  rows = []
  for uid in uniprot_ids:
      url = f"https://rest.uniprot.org/uniprotkb/{uid}.json"
      try:
          r = requests.get(url, timeout=30)
          if r.status_code == 200:
              data = r.json()
              pdbs = []
              for xref in data.get("uniProtKBCrossReferences", []):
                  if xref.get("database") == "PDB":
                      pdbs.append(xref["id"])
              rows.append({"uniprot": uid, "pdb_ids": ";".join(pdbs),
                           "best_pdb": pdbs[0] if pdbs else ""})
              print(f"{uid} -> {pdbs[:3]}")
          else:
              rows.append({"uniprot": uid, "pdb_ids": "", "best_pdb": ""})
              print(f"{uid} -> HTTP {r.status_code}")
      except Exception as e:
          rows.append({"uniprot": uid, "pdb_ids": "", "best_pdb": ""})
          print(f"{uid} -> ERROR {e}")
      time.sleep(0.5)  # be polite to the API

  with open(out, "w", newline="") as f:
      w = csv.DictWriter(f, fieldnames=["uniprot","pdb_ids","best_pdb"])
      w.writeheader(); w.writerows(rows)
  print(f"\nWrote {len(rows)} rows -> {out}")
  EOF

  @register_discoverer("af3_pla")
  def discover_af3_pla(raw_root):
      """
      AF3 PLA folders: summary_{UNIPROT}_{Nn}_{UNIPROT}_{LIG}_{rN}
      compound_key = {uniprot}_{lig}.lower()  e.g. b6ywb8_dtp
      Each folder has seed-1_sample-{0..4}/*_model.cif  (5 models)
      """
      all_records = defaultdict(list)

      for folder in sorted(raw_root.iterdir()):
          if not folder.is_dir() or folder.name.startswith("."):
              continue
          parts = folder.name.split("_")
          # PLA folders start with "summary"
          if len(parts) < 6 or parts[0] != "summary":
              continue

          uniprot = parts[1]          # e.g. B6YWB8
          lig     = parts[4]          # e.g. DTP
          pdb_key = f"{uniprot}_{lig}".lower()   # e.g. b6ywb8_dtp

          # Per-sample ranking scores from CSV
          ranking: dict[int, float] = {}
          ranking_csv = folder / f"{folder.name}_ranking_scores.csv"
          if ranking_csv.exists():
              try:
                  with open(ranking_csv, newline="") as f2:
                      for row in _csv.DictReader(f2):
                          ranking[int(row["sample"])] = float(row["ranking_score"])
              except Exception:
                  pass

          for sample_dir in sorted(folder.glob("seed-1_sample-*")):
              m = re.search(r"sample-(\d+)$", sample_dir.name)
              if not m:
                  continue
              sample_idx = int(m.group(1))
              cif_files = list(sample_dir.glob("*_model.cif"))
              if not cif_files:
                  continue
              conf_files = list(sample_dir.glob("*_summary_confidences.json"))
              raw: dict = {}
              confidence = ranking.get(sample_idx)
              if conf_files:
                  try:
                      meta = _json.loads(conf_files[0].read_text())
                      for k in ("ranking_score", "iptm", "ptm"):
                          if k in meta and isinstance(meta[k], (int, float)):
                              raw[k] = float(meta[k])
                      if confidence is None:
                          confidence = raw.get("ranking_score", raw.get("iptm"))
                  except Exception:
                      pass

              all_records[pdb_key].append(ModelRecord(
                  pdb_id=pdb_key, producer="af3_pla",
                  structure_path=cif_files[0],
                  raw_scores=raw, confidence=confidence,
              ))

      for pdb_key, records in sorted(all_records.items()):
          records.sort(key=lambda r: r.confidence or 0.0, reverse=True)
          yield pdb_key, records

  @register_discoverer("chai_pla")
  def discover_chai_pla(raw_root):
      """
      Chai PLA folders: {UNIPROT}_{Nn}__{PDB}__{LIG}  (double-underscore delim)
      compound_key = {uniprot}_{lig}.lower()  e.g. b6ywb8_dtp
      Each folder has pred.model_idx_{0..4}.cif  (5 models)
      Confidence from scores.model_idx_{N}.npz  (numpy array)
      """
      import numpy as _np

      all_records = defaultdict(list)

      for folder in sorted(raw_root.iterdir()):
          if not folder.is_dir() or folder.name.startswith("."):
              continue
          # Split on double underscore: ["{UNIPROT}_{Nn}", "{PDB}", "{LIG}"]
          dparts = folder.name.split("__")
          if len(dparts) != 3:
              continue
          uniprot_site = dparts[0]          # e.g. "B6YWB8_N1"
          pdb4         = dparts[1].upper()  # e.g. "6KBD"
          lig          = dparts[2]          # e.g. "DTP"
          uniprot      = uniprot_site.split("_")[0].upper()  # e.g. "B6YWB8"
          pdb_key      = f"{uniprot}_{lig}".lower()          # e.g. "b6ywb8_dtp"

          for idx in range(5):
              cif_path = folder / f"pred.model_idx_{idx}.cif"
              npz_path = folder / f"scores.model_idx_{idx}.npz"
              if not cif_path.exists():
                  continue

              confidence = None
              raw: dict = {}
              if npz_path.exists():
                  try:
                      npz = _np.load(str(npz_path))
                      # Common keys: aggregate_score, ptm, iptm, ...
                      for key in ("aggregate_score", "ptm", "iptm"):
                          if key in npz:
                              val = float(npz[key])
                              raw[key] = val
                              if key == "aggregate_score":
                                  confidence = val
                      if confidence is None and raw:
                          confidence = next(iter(raw.values()))
                  except Exception:
                      pass

              # Store pdb4 in raw_scores so pre-warm can find the reference
              raw["__pdb4__"] = pdb4

              all_records[pdb_key].append(ModelRecord(
                  pdb_id=pdb_key, producer="chai_pla",
                  structure_path=cif_path,
                  raw_scores=raw, confidence=confidence,
              ))

      for pdb_key, records in sorted(all_records.items()):
          records.sort(key=lambda r: r.confidence or 0.0, reverse=True)
          yield pdb_key, records

  # ── PLA reference pre-warm ────────────────────────────────────────────────
  import shutil as _shutil
  import requests as _preq
  import csv as _csv2
  import numpy as _np2

  def _prewarm_pla_refs(pla_raw_roots, refs_dir):
      """Pre-warm reference CIFs for PLA runs."""
      refs_dir = Path(refs_dir)
      refs_dir.mkdir(parents=True, exist_ok=True)

      # Load AF3 UniProt→PDB mapping
      umap = {}
      umap_path = BENCH_ROOT / "references" / "pla_uniprot_map.csv"
      if umap_path.exists():
          with open(umap_path) as f:
              for row in _csv2.DictReader(f):
                  umap[row["uniprot"].upper()] = row["best_pdb"].upper()

      pairs = []   # (compound_key, pdb4)
      seen  = set()

      for producer, raw_root in pla_raw_roots.items():
          raw_root = Path(raw_root)
          if not raw_root.is_dir():
              print(f"  SKIP {producer}: {raw_root} not found")
              continue

          for folder in sorted(raw_root.iterdir()):
              if not folder.is_dir() or folder.name.startswith("."):
                  continue

              if producer == "af3_pla":
                  parts = folder.name.split("_")
                  if len(parts) < 6 or parts[0] != "summary":
                      continue
                  uniprot = parts[1].upper()
                  lig     = parts[4]
                  pdb_key = f"{uniprot}_{lig}".lower()
                  pdb4    = umap.get(uniprot, "")
                  if not pdb4:
                      print(f"  WARNING: no PDB mapping for {uniprot} "
                            f"({pdb_key}) — add to pla_uniprot_map.csv")
                      continue

              elif producer == "chai_pla":
                  dparts = folder.name.split("__")
                  if len(dparts) != 3:
                      continue
                  uniprot = dparts[0].split("_")[0].upper()
                  pdb4    = dparts[1].upper()
                  lig     = dparts[2]
                  pdb_key = f"{uniprot}_{lig}".lower()

              else:
                  continue

              if pdb_key not in seen:
                  seen.add(pdb_key)
                  pairs.append((pdb_key, pdb4))

      done, from_local, from_rcsb, failed = [], [], [], []
      local_source = BENCH_ROOT / "references" / "ref_cifs"

      for compound_key, pdb4 in pairs:
          dest = refs_dir / f"{compound_key}.cif"
          dest_gz = refs_dir / f"{compound_key}.cif.gz"
          if dest.exists() or dest_gz.exists():
              done.append(compound_key)
              continue

          # Try local refs_dir first (by PDB code)
          for local_name in (f"{pdb4}.cif", f"{pdb4.lower()}.cif",
                              f"{pdb4}.cif.gz", f"{pdb4.lower()}.cif.gz",
                              f"{pdb4}-assembly1.cif",
                              f"{pdb4}-assembly1.cif.gz"):
              local = local_source / local_name
              if local.exists():
                  _shutil.copyfile(local, dest_gz if local_name.endswith(".gz") else dest)
                  from_local.append(compound_key)
                  done.append(compound_key)
                  break
          else:
              # Download from RCSB
              data = None
              for url in [
                  f"https://files.rcsb.org/download/{pdb4}-assembly1.cif.gz",
                  f"https://files.rcsb.org/download/{pdb4}.cif.gz",
              ]:
                  try:
                      r = _preq.get(url, timeout=60)
                      if r.status_code == 200:
                          data = r.content; break
                  except Exception:
                      pass
              if data:
                  dest_gz.write_bytes(data)
                  from_rcsb.append(compound_key)
                  done.append(compound_key)
                  print(f"  downloaded {pdb4} -> {compound_key}")
              else:
                  failed.append(compound_key)
                  print(f"  FAILED {pdb4} ({compound_key})")

      print(f"\nPLA ref pre-warm: {len(done)} ok "
            f"({len(from_local)} local, {len(from_rcsb)} RCSB), "
            f"{len(failed)} failed")
      if failed:
          print("Failed keys:", failed[:10])
      return done, failed

SECTION 12: EXPORT PLA CSVs (4c equivalent)
================================================================================

Add a new cell for PLA CSV export:

  # ── PLA Export ────────────────────────────────────────────────────────────
  try:
      _pla_src = pla_df.copy()
  except NameError:
      import pandas as pd
      _pla_src = None
      for _c in [PLA_OUTPUT_DIR / "benchmark_with_pocket_rmsd.parquet",
                 PLA_OUTPUT_DIR / "benchmark.parquet"]:
          if _c.exists():
              _pla_src = (pd.read_parquet(_c) if _c.suffix == ".parquet"
                          else pd.read_csv(_c))
              print(f"Loaded {_c.name} ({len(_pla_src)} rows)")
              break
      if _pla_src is None:
          raise FileNotFoundError("Run PLA scoring + pocket RMSD first.")

  _PLA_RENAME = {
      "bisy_rmsd":   "pose rmsd",
      "pocket_rmsd": "pocket rmsd",
      "qs_global":   "qs score",
      "lddt_pli":    "lddt-pli",
  }
  _PLA_METRIC_COLS = ["bisy_rmsd", "pocket_rmsd", "qs_global", "lddt_pli"]

  def _pla_id(compound_key):
      # compound_key = "b6ywb8_dtp"  →  id = "b6ywb8"  (UniProt, lowercase)
      return compound_key.split("_")[0].lower()

  pla_ok = _pla_src[_pla_src["status"] == "ok"].copy()
  pla_ok["id"] = pla_ok["pdb_id"].apply(_pla_id)

  PLA_EVAL_DIR.mkdir(parents=True, exist_ok=True)

  for producer, grp in pla_ok.groupby("producer"):
      # Best: one row per UniProt ID, lowest pose RMSD
      best_idx = (
          grp.dropna(subset=["bisy_rmsd"])
             .groupby("id")["bisy_rmsd"]
             .idxmin().dropna().astype(int)
      )
      best = grp.loc[best_idx].copy().rename(columns=_PLA_RENAME)
      best_cols = (["id"]
                   + [_PLA_RENAME[c] for c in _PLA_METRIC_COLS if c in grp.columns]
                   + ["confidence"])
      best[[c for c in best_cols if c in best.columns]].sort_values("id").to_csv(
          PLA_EVAL_DIR / f"{producer}mainligand_best.csv", index=False)

      # Avg: mean of all metrics per UniProt ID
      avg_cols = [c for c in _PLA_METRIC_COLS if c in grp.columns]
      avg = (grp.groupby("id")[avg_cols + ["confidence"]]
               .mean().rename(columns=_PLA_RENAME).reset_index())
      avg_out = ["id"] + [_PLA_RENAME[c] for c in avg_cols] + ["confidence"]
      avg[[c for c in avg_out if c in avg.columns]].sort_values("id").to_csv(
          PLA_EVAL_DIR / f"{producer}mainligand_avg.csv", index=False)

      print(f"{producer}: {len(best_idx)} best-rows, {len(avg)} avg-rows")

  print("\nPLA CSVs written to:", PLA_EVAL_DIR)
